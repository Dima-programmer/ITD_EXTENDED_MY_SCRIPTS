(function() {
    'use strict';

    // Function to hide sections with class "suggestions", specific .explore-section, and section.top-clans
    function hideElements() {
        // Hide all section.suggestions
        const suggestions = document.querySelectorAll('section.suggestions');
        suggestions.forEach(section => {
            section.style.display = 'none';
        });

        // Hide .explore-section that contain h3 with specific text
        const exploreSections = document.querySelectorAll('.explore-section');
        exploreSections.forEach(element => {
            const h3 = element.querySelector('h3');
            if (h3 && (h3.textContent.trim() === 'Кого читать' || h3.textContent.trim() === 'Топ кланов')) {
                element.style.display = 'none';
            }
        });

        // Hide all section.top-clans
        const topClans = document.querySelectorAll('section.top-clans');
        topClans.forEach(section => {
            section.style.display = 'none';
        });
    }

    // Hide on page load
    hideElements();

    // Observe for dynamically added elements
    const observer = new MutationObserver(hideElements);
    observer.observe(document.body, { childList: true, subtree: true });
})();
