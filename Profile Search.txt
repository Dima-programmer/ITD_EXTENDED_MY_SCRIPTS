(function() {
    'use strict';

    let currentSearch = null; // To track the current search process
    let hideNewPostsObserver = null; // Observer to hide new posts during search
    let postsCache = new Map(); // Global cache for posts to avoid re-scraping
    let cacheLoaded = false; // Flag to check if cache is loaded

    async function scrapePosts(searchQuery) {
        // Cancel any ongoing search
        if (currentSearch) {
            currentSearch.cancelled = true;
            if (hideNewPostsObserver) {
                hideNewPostsObserver.disconnect();
                hideNewPostsObserver = null;
            }
        }
        currentSearch = { cancelled: false };

        const query = searchQuery.trim().toLowerCase();

        // If query is empty, show all posts immediately without scraping
        if (query === '') {
            const postContainers = document.querySelectorAll('[class*="post-container"]');
            postContainers.forEach(container => {
                container.style.display = ''; // Show all
            });

            const postList = document.querySelector('[class*="post-list"]');
            if (postList) {
                // Remove spinner or message
                const existingSpinner = postList.querySelector('.search-spinner');
                if (existingSpinner) {
                    existingSpinner.remove();
                }
                const existingMessage = postList.querySelector('.no-posts-message');
                if (existingMessage) {
                    existingMessage.remove();
                }
            }
            currentSearch = null;
            return;
        }

        // Получаем USERNAME из текущей ссылки (например, сайт.com/USERNAME)
        const pathname = window.location.pathname;
        const USERNAME = pathname.split('/')[1]; // Предполагаем, что URL выглядит как /USERNAME

        if (!USERNAME) {
            console.error('Не удалось получить USERNAME из URL');
            return;
        }

        const CONFIG = {
            limit: 50,
            maxEmpty: 3
        };

        // Show spinner and hide posts at the start
        const postContainers = document.querySelectorAll('[class*="post-container"]');
        postContainers.forEach(container => {
            container.style.display = 'none';
        });

        const postList = document.querySelector('[class*="post-list"]');
        if (postList) {
            // Remove existing message or spinner
            const existingMessage = postList.querySelector('.no-posts-message');
            if (existingMessage) {
                existingMessage.remove();
            }
            const existingSpinner = postList.querySelector('.search-spinner');
            if (existingSpinner) {
                existingSpinner.remove();
            }

            // Add spinner
            const spinner = document.createElement('div');
            spinner.className = 'search-spinner';
            spinner.innerHTML = '<svg class="loader__spinner svelte-lrcio1" viewBox="0 0 24 24" fill="none" xmlns="http://www.w3.org/2000/svg" width="24" height="24"><circle class="loader__track svelte-lrcio1" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="3"></circle><path class="loader__arc svelte-lrcio1" d="M12 2a10 10 0 0 1 10 10" stroke="currentColor" stroke-width="3" stroke-linecap="round"></path></svg>';
            spinner.style.cssText = `
                display: flex;
                justify-content: center;
                align-items: center;
                padding: 20px;
                color: #666;
                animation: spin 1s linear infinite;
            `;
            postList.appendChild(spinner);
        }

        // Start observing for new posts and hide them
        hideNewPostsObserver = new MutationObserver(function(mutations) {
            mutations.forEach(function(mutation) {
                if (mutation.type === 'childList') {
                    mutation.addedNodes.forEach(function(node) {
                        if (node.nodeType === Node.ELEMENT_NODE && node.matches && node.matches('[class*="post-container"]')) {
                            node.style.display = 'none';
                        }
                        // Also check descendants
                        const newPosts = node.querySelectorAll ? node.querySelectorAll('[class*="post-container"]') : [];
                        newPosts.forEach(container => {
                            container.style.display = 'none';
                        });
                    });
                }
            });
        });
        hideNewPostsObserver.observe(document.body, {
            childList: true,
            subtree: true
        });

        async function getNewToken() {
            const url = 'https://xn--d1ah4a.com/api/v1/auth/refresh';
            const options = { method: 'POST' };
            try {
                const res = await fetch(url, options);
                const data = await res.json();
                return data.accessToken || data.token;
            } catch (e) {
                console.error('Ошибка при получении токена:', e);
                return null;
            }
        }

        let token = await getNewToken();
        if (!token) {
            console.error('Не удалось получить токен');
            return;
        }

        // Use cache if available, else scrape
        let allPosts = new Map(postsCache);
        if (!cacheLoaded) {
            let isRunning = true;
            let currentDate = new Date(); // Начинаем с сегодняшней даты
            let emptyHits = 0;

            // Однопоточный скрапинг (optimized: reduced delay)
            while (isRunning && emptyHits < CONFIG.maxEmpty && !currentSearch.cancelled) {
                try {
                    const cursor = encodeURIComponent(currentDate.toISOString());
                    const url = `https://xn--d1ah4a.com/api/posts/user/${USERNAME}?limit=${CONFIG.limit}&cursor=${cursor}&sort=new`;
                    const options = { headers: { 'Authorization': `Bearer ${token}` } };
                    const res = await fetch(url, options);

                    if (res.status === 401) {
                        token = await getNewToken();
                        if (!token) break;
                        continue;
                    }

                    const result = await res.json();

                    const posts = result.data?.posts || [];

                    if (posts.length > 0) {
                        for (let i = 0; i < posts.length; i++) {
                            allPosts.set(posts[i].id, posts[i]);
                        }
                        emptyHits = 0;
                        // Двигаемся в сторону устаревания: вычитаем 1 день
                        currentDate.setDate(currentDate.getDate() - 1);
                    } else {
                        emptyHits++;
                    }

                    if (posts.length < CONFIG.limit) break;

                } catch (e) {
                    console.error('Ошибка при скрапинге:', e);
                    emptyHits++;
                    await new Promise(r => setTimeout(r, 200)); // Reduced delay
                }
            }

            if (currentSearch.cancelled) {
                if (hideNewPostsObserver) {
                    hideNewPostsObserver.disconnect();
                    hideNewPostsObserver = null;
                }
                return;
            }

            isRunning = false;
            postsCache = new Map(allPosts); // Cache the posts
            cacheLoaded = true;
        }

        const finalData = Array.from(allPosts.values());

        // Фильтрация по запросу (optimized: use includes for speed)
        let filteredData = finalData.filter(post => {
            const content = (post.content || post.text || '').toLowerCase();
            return content.includes(query);
        });

        // Создаём Set с id отфильтрованных постов
        const filteredIds = new Set(filteredData.map(post => post.id));

        // Теперь ждём, пока все filteredIds появятся на странице, провоцируя загрузку путём движения sentinel
        function checkAllPostsLoaded() {
            if (!currentSearch || currentSearch.cancelled) {
                if (hideNewPostsObserver) {
                    hideNewPostsObserver.disconnect();
                    hideNewPostsObserver = null;
                }
                return;
            }

            const currentPostContainers = document.querySelectorAll('[class*="post-container"]');
            const currentIds = new Set();
            currentPostContainers.forEach(container => {
                const postId = container.getAttribute('data-post-id');
                if (postId) currentIds.add(postId);
            });

            // Проверяем, все ли filteredIds присутствуют
            const allLoaded = Array.from(filteredIds).every(id => currentIds.has(id));

            if (allLoaded) {
                // Все посты загружены, завершаем поиск
                completeSearch(filteredIds);
            } else {
                // Провоцируем загрузку новых постов путём движения sentinel элемента (optimized: faster movement)
                const sentinel = document.querySelector('[class*="post-list__sentinel"]');
                if (sentinel) {
                    const originalPosition = sentinel.style.position || '';
                    const originalTop = sentinel.style.top || '';
                    sentinel.style.position = 'relative';
                    sentinel.style.top = '-500px'; // Move out of viewport to trigger intersection
                    setTimeout(() => {
                        sentinel.style.top = originalTop; // Move back immediately
                        sentinel.style.position = originalPosition;
                    }, 1); // Reduced timeout
                }
                // Продолжаем ждать, проверяем снова через меньшее время
                setTimeout(checkAllPostsLoaded, 10); // Reduced delay to 10ms
            }
        }

        function completeSearch(filteredIds) {
            if (!currentSearch || currentSearch.cancelled) {
                if (hideNewPostsObserver) {
                    hideNewPostsObserver.disconnect();
                    hideNewPostsObserver = null;
                }
                return;
            }

            // Disconnect the observer
            if (hideNewPostsObserver) {
                hideNewPostsObserver.disconnect();
                hideNewPostsObserver = null;
            }

            // Находим все div с классом post-container и фильтруем по data-post-id
            const postContainers = document.querySelectorAll('[class*="post-container"]');
            postContainers.forEach(container => {
                const postId = container.getAttribute('data-post-id');
                if (postId && filteredIds.has(postId)) {
                    container.style.display = ''; // Показать
                } else {
                    container.style.display = 'none'; // Скрыть
                }
            });

            // Находим блок post-list и добавляем/убираем надпись "посты не найдены" или spinner
            if (postList) {
                // Убираем spinner
                const existingSpinner = postList.querySelector('.search-spinner');
                if (existingSpinner) {
                    existingSpinner.remove();
                }

                // Убираем существующую надпись, если есть
                const existingMessage = postList.querySelector('.no-posts-message');
                if (existingMessage) {
                    existingMessage.remove();
                }

                if (filteredData.length === 0) {
                    // Добавляем надпись "посты не найдены"
                    const noPostsMessage = document.createElement('div');
                    noPostsMessage.className = 'no-posts-message';
                    noPostsMessage.textContent = 'посты не найдены';
                    noPostsMessage.style.cssText = `
                        text-align: center;
                        padding: 20px;
                        font-size: 18px;
                        color: #666;
                    `;
                    postList.appendChild(noPostsMessage);
                }
            }

            // Извлекаем id и content, сохраняем в формате id:content
            const postsData = filteredData.map(post => `${post.id}:${post.content || post.text || 'Нет содержимого'}`);
            console.log(`Отфильтрованные посты по запросу "${query}":`, postsData);

            currentSearch = null;
        }

        // Начинаем проверку
        checkAllPostsLoaded();
    }

    async function addSearchBar() {
        // Check if the search bar is already added to avoid duplicates
        if (document.querySelector('.custom-search-container')) return;

        // Find the profile-tabs element (using class contains in case of multiple classes)
        const profileTabs = document.querySelector('[class*="profile-tabs"]');
        if (!profileTabs) return; // Exit if not found

        // Find the parent element with class profile-card (using closest for upward search)
        const profileCard = profileTabs.closest('[class*="profile-card"]');
        if (!profileCard) return; // Exit if not found

        // Create the search bar container
        const searchContainer = document.createElement('div');
        searchContainer.className = 'custom-search-container'; // Add a class to prevent duplicates
        searchContainer.style.cssText = `
            position: relative;
            display: flex;
            justify-content: center;
            align-items: center;
            width: 98%;
            margin: 10px auto;
            padding: 10px;
            background: inherit; /* Inherit background from parent (site's background) */
            border-radius: 25px; /* Strongly rounded edges */
            box-shadow: 0 2px 5px rgba(128,128,128,0.1);
        `;

        // Create the input field
        const searchInput = document.createElement('input');
        searchInput.type = 'text';
        searchInput.placeholder = 'Search...';
        searchInput.style.cssText = `
            flex: 1;
            padding: 10px 50px 10px 40px; /* Left padding for icon, right for clear button */
            border: none;
            outline: none;
            background: transparent; /* Transparent background */
            font-size: 16px;
            border-radius: 20px; /* Rounded edges */
        `;

        // Create the search icon (larger and positioned at the start, slightly lowered)
        const searchIcon = document.createElement('div');
        searchIcon.innerHTML = '<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="none" width="24" height="24"><path stroke="currentColor" stroke-linecap="round" stroke-linejoin="round" stroke-width="3" d="m19.5 19.5-3-3M11 4.5a6.5 6.5 0 1 1 0 13 6.5 6.5 0 0 1 0-13Z"></path></svg>';
        searchIcon.style.cssText = `
            position: absolute;
            left: 10px; /* Position at the start inside the input */
            top: 50%;
            transform: translateY(-40%); /* Slightly lowered */
            pointer-events: none;
            color: #666; /* Icon color */
        `;

        // Create the clear button (circular, transparent background, at the end, moved further left)
        const clearButton = document.createElement('button');
        clearButton.innerHTML = '<svg xmlns="http://www.w3.org/2000/svg" height="24px" viewBox="0 -960 960 960" width="24px" fill="#1f1f1f"><path d="m456-320 104-104 104 104 56-56-104-104 104-104-56-56-104 104-104-104-56 56 104 104-104 104 56 56Zm-96 160q-19 0-36-8.5T296-192L80-480l216-288q11-15 28-23.5t36-8.5h440q33 0 56.5 23.5T880-720v480q0 33-23.5 56.5T800-160H360ZM180-480l180 240h440v-480H360L180-480Zm400 0Z"/></svg>';
        clearButton.style.cssText = `
            position: absolute;
            right: 0px; /* Moved further left */
            top: 50%;
            transform: translateY(-50%);
            width: 30px;
            height: 30px;
            border: none;
            border-radius: 50%; /* Circular */
            background: inherit; /* Transparent background */
            cursor: pointer;
            display: flex;
            justify-content: center;
            align-items: center;
            color: #666;
        `;
        clearButton.addEventListener('click', () => {
            searchInput.value = '';
            searchInput.focus(); // Focus the input after clearing
            searchInput.dispatchEvent(new Event('input')); // Trigger the input event to perform search
        });

        // Debounce timer for automatic search (optimized: reduced debounce)
        let debounceTimer;

        // Add event listener for automatic search on input change with debounce
        searchInput.addEventListener('input', () => {
            clearTimeout(debounceTimer);
            debounceTimer = setTimeout(async () => {
                await scrapePosts(searchInput.value);
            }, 150); // Reduced debounce
        });

        // Optional: Keep Enter key for immediate search
        searchInput.addEventListener('keydown', async (e) => {
            if (e.key === 'Enter') {
                clearTimeout(debounceTimer);
                await scrapePosts(searchInput.value);
            }
        });

        // Append input, icon, and clear button to container
        searchContainer.appendChild(searchInput);
        searchContainer.appendChild(searchIcon);
        searchContainer.appendChild(clearButton);

        // Insert the search container before the profile-tabs element
        profileTabs.parentNode.insertBefore(searchContainer, profileTabs);

        // Initial search to show all posts
        await scrapePosts('');
    }

    // Add CSS for spinner animation
    const style = document.createElement('style');
    style.textContent = `
        @keyframes spin {
            0% { transform: rotate(0deg); }
            100% { transform: rotate(360deg); }
        }
        .search-spinner svg {
            animation: spin 1s linear infinite;
        }
    `;
    document.head.appendChild(style);

    // Run on page load
    window.addEventListener('load', addSearchBar);

    // Also observe for dynamic content changes
    const observer = new MutationObserver(function(mutations) {
        mutations.forEach(function(mutation) {
            if (mutation.type === 'childList') {
                addSearchBar();
            }
        });
    });

    // Start observing the document body for changes
    observer.observe(document.body, {
        childList: true,
        subtree: true
    });
})();
